```typescript
import { Router, Request, Response, NextFunction } from 'express';
import axios from 'axios';
import { authenticate } from '../middleware/auth.js';
import { AppError } from '../middleware/errorHandler.js';

const router = Router();

/** Parse da chave NF-e 44 dígitos (Modelo 55 e 57/CT-e) */
function parseChaveNFe(chave: string) {
  const digits = chave.replace(/\D/g, '');
  if (digits.length !== 44) return null;

  const cUF = digits.slice(0, 2);
  const aamm = digits.slice(2, 6);
  const cnpjEmit = digits.slice(6, 20);
  const mod = digits.slice(20, 22);
  const serie = digits.slice(22, 25);
  const nnf = digits.slice(25, 34);

  const year = parseInt(aamm.slice(0, 2), 10);
  const month = aamm.slice(2, 4);
  const fullYear = year >= 90 ? `19${aamm.slice(0, 2)}` : `20${aamm.slice(0, 2)}`;
  const emissionDate = `${fullYear}-${month}-01`;

  const ufMap: Record<string, string> = {
    '11': 'RO', '12': 'AC', '13': 'AM', '14': 'RR', '15': 'PA',
    '16': 'AP', '17': 'TO', '21': 'MA', '22': 'PI', '23': 'CE',
    '24': 'RN', '25': 'PB', '26': 'PE', '27': 'AL', '28': 'SE',
    '29': 'BA', '31': 'MG', '32': 'ES', '33': 'RJ', '35': 'SP',
    '41': 'PR', '42': 'SC', '43': 'RS', '50': 'MS', '51': 'MT',
    '52': 'GO', '53': 'DF',
  };

  return {
    chave: digits,
    cUF,
    uf: ufMap[cUF] ?? cUF,
    aamm,
    cnpjEmit,
    mod,
    serie: String(parseInt(serie, 10)),
    nnf: String(parseInt(nnf, 10)),
    emissionDate,
  };
}

/** Extrai a chave de 44 dígitos de uma URL de QR Code de NF-e */
function extractChaveFromUrl(input: string): string {
  const digits = input.replace(/\D/g, '');
  if (digits.length === 44) return digits;

  // QR Code NF-e contém URL com param chNFe ou p=CHAVE|...
  try {
    const url = new URL(input);
    const chNFe = url.searchParams.get('chNFe') ?? url.searchParams.get('p');
    if (chNFe) {
      const extracted = chNFe.split('|')[0].replace(/\D/g, '');
      if (extracted.length === 44) return extracted;
    }
  } catch {
    // não é URL, continua
  }
  return digits;
}

// GET /api/nfe/:chave
router.get('/:chave', authenticate, async (req: Request, res: Response, next: NextFunction) => {
  try {
    const raw = decodeURIComponent(req.params.chave);
    const chave = extractChaveFromUrl(raw);

    if (!chave || chave.length !== 44) {
      throw new AppError('Chave NF-e inválida (deve ter 44 dígitos)', 400);
    }

    const parsed = parseChaveNFe(chave);
    if (!parsed) {
      throw new AppError('Não foi possível interpretar a chave NF-e', 400);
    }

    // Busca razão social do emitente via BrasilAPI (gratuita, sem autenticação)
    let emitenteRazaoSocial: string | null = null;
    let emitenteTradeName: string | null = null;
    try {
      const cnpjRes = await axios.get(
        `https://brasilapi.com.br/api/cnpj/v1/${parsed.cnpjEmit}`,
        { timeout: 5000 }
      );
      emitenteRazaoSocial = cnpjRes.data?.razao_social ?? null;
      emitenteTradeName = cnpjRes.data?.nome_fantasia ?? null;
    } catch {
      // BrasilAPI falhou — retorna sem razão social, não bloqueia operação
    }

    res.json({
      success: true,
      data: {
        chave: parsed.chave,
        uf: parsed.uf,
        serie: parsed.serie,
        numero: parsed.nnf,
        emissionDate: parsed.emissionDate,
        cnpjEmitente: parsed.cnpjEmit,
        emitenteRazaoSocial,
        emitenteTradeName,
        modelo: parsed.mod === '55' ? 'NF-e' : parsed.mod === '57' ? 'CT-e' : parsed.mod,
      },
    });
  } catch (err) {
    next(err);
  }
});

export { router as nfeRouter };
```

```typescript
import React, { useEffect, useRef, useState } from 'react';
import { Html5Qrcode } from 'html5-qrcode';

interface QRCodeScannerProps {
  onScan: (result: string) => void;
  onClose: () => void;
}

const QRCodeScanner: React.FC<QRCodeScannerProps> = ({ onScan, onClose }) => {
  const scannerRef = useRef<Html5Qrcode | null>(null);
  const containerId = 'qr-scanner-container';
  const [error, setError] = useState<string | null>(null);
  const [started, setStarted] = useState(false);

  useEffect(() => {
    const scanner = new Html5Qrcode(containerId);
    scannerRef.current = scanner;

    scanner
      .start(
        { facingMode: 'environment' }, // câmera traseira
        { fps: 10, qrbox: { width: 250, height: 250 } },
        (decodedText) => {
          scanner.stop().catch(() => {});
          onScan(decodedText);
        },
        () => {} // erros de frame são normais, ignorar
      )
      .then(() => setStarted(true))
      .catch((err) => {
        setError('Não foi possível acessar a câmera. Verifique as permissões.');
        console.error('QR scanner error:', err);
      });

    return () => {
      if (scannerRef.current && started) {
        scannerRef.current.stop().catch(() => {});
      }
    };
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  return (
    <div className="fixed inset-0 z-50 bg-black/80 flex flex-col items-center justify-center p-4">
      <div className="bg-white dark:bg-slate-800 rounded-2xl overflow-hidden w-full max-w-sm">
        {/* Header */}
        <div className="flex items-center justify-between p-4 border-b border-slate-200 dark:border-slate-700">
          <h3 className="font-semibold text-slate-900 dark:text-white">
            Escanear QR Code da DANFE
          </h3>
          <button
            type="button"
            onClick={onClose}
            className="text-slate-400 hover:text-slate-600 p-1 rounded-lg"
          >
            <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        </div>

        {/* Scanner area */}
        <div className="p-4">
          {error ? (
            <div className="text-center py-8">
              <p className="text-red-600 text-sm mb-4">{error}</p>
              <button
                type="button"
                onClick={onClose}
                className="px-4 py-2 bg-slate-100 dark:bg-slate-700 rounded-lg text-sm font-medium text-slate-700 dark:text-slate-300"
              >
                Fechar
              </button>
            </div>
          ) : (
            <div id={containerId} className="w-full rounded-lg overflow-hidden" />
          )}
        </div>

        <div className="px-4 pb-4 text-center">
          <p className="text-xs text-slate-500 dark:text-slate-400">
            Aponte a câmera para o QR Code impresso na DANFE
          </p>
        </div>
      </div>
    </div>
  );
};

export default QRCodeScanner;
```

```typescript
import { useState, useCallback } from 'react';
import { nfeApi } from '../lib/api';

export interface NfeLookupResult {
  chave: string;
  uf: string;
  serie: string;
  numero: string;
  emissionDate: string; // YYYY-MM-DD
  cnpjEmitente: string;
  emitenteRazaoSocial: string | null;
  emitenteTradeName: string | null;
  modelo: string;
}

interface UseNfeLookupReturn {
  lookup: (chaveOrUrl: string) => Promise<NfeLookupResult | null>;
  loading: boolean;
  error: string | null;
  clearError: () => void;
}

/** Extrai a chave de 44 dígitos de uma string (URL de QR Code ou dígitos diretos) */
function extractChave(input: string): string {
  const digits = input.replace(/\D/g, '');
  if (digits.length === 44) return digits;

  try {
    const url = new URL(input);
    const chNFe = url.searchParams.get('chNFe') ?? url.searchParams.get('p');
    if (chNFe) {
      const extracted = chNFe.split('|')[0].replace(/\D/g, '');
      if (extracted.length === 44) return extracted;
    }
  } catch {
    // não é URL
  }
  return digits;
}

export function useNfeLookup(): UseNfeLookupReturn {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const lookup = useCallback(async (chaveOrUrl: string): Promise<NfeLookupResult | null> => {
    const chave = extractChave(chaveOrUrl);

    if (chave.length !== 44) {
      setError('Chave NF-e deve ter 44 dígitos.');
      return null;
    }

    setLoading(true);
    setError(null);
    try {
      const res = await nfeApi.lookup(chave);
      return res.data.data;
    } catch (err: any) {
      const msg = err.response?.data?.error ?? err.message ?? 'Erro ao consultar NF-e';
      setError(msg);
      return null;
    } finally {
      setLoading(false);
    }
  }, []);

  const clearError = useCallback(() => setError(null), []);

  return { lookup, loading, error, clearError };
}
```

                  <option value="">Selecione o veículo</option>
                  {vehicleTypes.map(vt => (
                    <option key={vt.id} value={vt.id}>
                      {vt.name} — {vt.maxWeight}kg / {vt.maxVolume}m³
                    </option>
                  ))}
                </select>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1">
                  Tipo de frete
                </label>
                <div className="flex gap-2">
                  {(['COMPLEMENTO', 'FECHADO'] as const).map(t => (
                    <button
                      key={t}
                      type="button"
                      onClick={() => setManifestForm(prev => ({ ...prev, type: t }))}
                      className={`flex-1 py-2.5 rounded-lg border text-sm font-medium transition-colors ${
                        manifestForm.type === t
                          ? 'border-indigo-600 bg-indigo-50 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400 dark:border-indigo-500'
                          : 'border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300'
                      }`}
                    >
                      {t === 'COMPLEMENTO' ? 'Complemento' : 'Carga Fechada'}
                    </button>
                  ))}
                </div>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1">
                  Data prevista de saída
                </label>
                <input
                  type="date"
                  value={manifestForm.scheduledDate}
                  onChange={e => setManifestForm(prev => ({ ...prev, scheduledDate: e.target.value }))}
                  className="w-full px-3 py-2.5 rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-700 text-slate-900 dark:text-white text-sm"
                />
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1">
                  Observações
                </label>
                <textarea
                  rows={2}
                  value={manifestForm.notes}
                  onChange={e => setManifestForm(prev => ({ ...prev, notes: e.target.value }))}
                  className="w-full px-3 py-2.5 rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-700 text-slate-900 dark:text-white text-sm resize-none"
                />
              </div>

              <button
                type="button"
                onClick={handleCreateManifest}
                disabled={manifestSaving}
                className="w-full py-3 bg-indigo-600 hover:bg-indigo-700 disabled:opacity-50 text-white rounded-xl font-semibold text-sm transition-colors"
              >
                {manifestSaving ? 'Criando...' : 'Criar Manifesto'}
              </button>
            </div>
          </div>
        </div>
      )}

      <BottomNav items={navItems} active={tab} onChange={(k) => setTab(k as Tab)} />
    </div>
  );
};

export default HubHome;

## Task 9: Frontend — LinkHome Redesign Mobile-First

**Files:**
- Reescrever: `components/operacao/LinkHome.tsx`

- [ ] **Passo 1: Reescrever `components/operacao/LinkHome.tsx`**

```typescript
import React, { useState, useEffect, useCallback } from 'react';
import { useAuthStore } from '../../store/authStore';
import { manifestsApi } from '../../lib/api';
import type { Manifest } from '../../types/index';
import BottomNav, { type BottomNavItem } from './BottomNav';

type Tab = 'ofertas' | 'viagens' | 'historico';

const statusManifestLabel: Record<string, string> = {
  PENDING_ACCEPT: 'Aguardando aceite',
  ACCEPTED: 'Aceito',
  IN_PROGRESS: 'Em andamento',
  COMPLETED: 'Concluído',
  DRAFT: 'Rascunho',
};

const LinkHome: React.FC = () => {
  const { user, logout } = useAuthStore();
  const [tab, setTab] = useState<Tab>('ofertas');
  const [manifests, setManifests] = useState<Manifest[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [actionLoading, setActionLoading] = useState<string | null>(null);

  const load = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const res = await manifestsApi.list({});
      const all = ((res as any).data ?? res) as Manifest[];
      setManifests(Array.isArray(all) ? all : []);
    } catch (e: any) {
      setError(e.response?.data?.error || e.message || 'Erro ao carregar');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { load(); }, [load]);

  const ofertas = manifests.filter(m => m.status === 'PENDING_ACCEPT');
  const viagens = manifests.filter(m => ['ACCEPTED', 'IN_PROGRESS'].includes(m.status));
  const historico = manifests.filter(m => m.status === 'COMPLETED');

  const handleAccept = async (id: string) => {
    setActionLoading(id);
    try {
      await manifestsApi.accept(id);
      await load();
    } catch (e: any) {
      alert(e.response?.data?.error || e.message);
    } finally {
      setActionLoading(null);
    }
  };

  const handleReject = async (id: string) => {
    const reason = window.prompt('Motivo da recusa (opcional):') ?? '';
    setActionLoading(id);
    try {
      await manifestsApi.reject(id, reason);
      await load();
    } catch (e: any) {
      alert(e.response?.data?.error || e.message);
    } finally {
      setActionLoading(null);
    }
  };

  const handleUpdateStatus = async (id: string, status: string) => {
    setActionLoading(`${id}-${status}`);
    try {
      await manifestsApi.updateStatus(id, status);
      await load();
    } catch (e: any) {
      alert(e.response?.data?.error || e.message);
    } finally {
      setActionLoading(null);
    }
  };

  const ManifestCard: React.FC<{ manifest: Manifest; showActions?: boolean }> = ({ manifest, showActions }) => {
    const m = manifest as any;
    const isLoading = actionLoading?.startsWith(manifest.id);
    const totalWeight = m.totalWeight ?? '—';
    const totalVolume = m.totalVolume ?? '—';
    const demandsCount = m._count?.demands ?? m.demands?.length ?? 0;

    return (
      <div className="bg-white dark:bg-slate-800 rounded-2xl shadow-sm overflow-hidden">
        {/* Cabeçalho do card */}
        <div className="p-4">
          <div className="flex items-start justify-between gap-2 mb-3">
            <div>
              <p className="text-sm font-semibold text-slate-900 dark:text-white">
                {m.origin ?? 'Origem N/D'} → {m.destination ?? 'Destino N/D'}
              </p>
              <p className="text-xs text-slate-500 dark:text-slate-400 mt-0.5">
                {demandsCount} pedido{demandsCount !== 1 ? 's' : ''}
                {totalWeight !== '—' ? ` • ${totalWeight}kg` : ''}
                {totalVolume !== '—' ? ` • ${totalVolume}m³` : ''}
              </p>
            </div>
            {m.scheduledDate && (
              <p className="text-xs text-slate-500 dark:text-slate-400 shrink-0">
                {new Date(m.scheduledDate).toLocaleDateString('pt-BR')}
              </p>
            )}
          </div>

          {m.vehicleType && (
            <div className="flex items-center gap-1.5 text-xs text-slate-500 dark:text-slate-400 mb-3">
              <svg className="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
                  d="M8 7h12m0 0l-4-4m4 4l-4 4m0 6H4m0 0l4 4m-4-4l4-4"/>
              </svg>
              {m.vehicleType.name}
            </div>
          )}

          {/* Ações */}
          {showActions && manifest.status === 'PENDING_ACCEPT' && (
            <div className="flex gap-2">
              <button
                type="button"
                onClick={() => handleReject(manifest.id)}
                disabled={isLoading}
                className="flex-1 py-2.5 rounded-xl border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 text-sm font-medium disabled:opacity-50"
              >
                Recusar
              </button>
              <button
                type="button"
                onClick={() => handleAccept(manifest.id)}
                disabled={isLoading}
                className="flex-2 flex-grow py-2.5 rounded-xl bg-indigo-600 hover:bg-indigo-700 text-white text-sm font-semibold disabled:opacity-50 transition-colors"
              >
                {actionLoading === manifest.id ? 'Aceitando...' : 'Aceitar Carga'}
              </button>
            </div>
          )}

          {showActions && manifest.status === 'ACCEPTED' && (
            <button
              type="button"
              onClick={() => handleUpdateStatus(manifest.id, 'IN_PROGRESS')}
              disabled={!!actionLoading}
              className="w-full py-2.5 rounded-xl bg-green-600 hover:bg-green-700 text-white text-sm font-semibold disabled:opacity-50 transition-colors"
            >
              {actionLoading === `${manifest.id}-IN_PROGRESS` ? 'Atualizando...' : 'Iniciar Viagem'}
            </button>
          )}

          {showActions && manifest.status === 'IN_PROGRESS' && (
            <button
              type="button"
              onClick={() => {
                if (window.confirm('Confirmar entrega de toda a carga deste manifesto?')) {
                  handleUpdateStatus(manifest.id, 'COMPLETED');
                }
              }}
              disabled={!!actionLoading}
              className="w-full py-2.5 rounded-xl bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-semibold disabled:opacity-50 transition-colors"
            >
              {actionLoading === `${manifest.id}-COMPLETED` ? 'Finalizando...' : 'Concluir Entrega'}
            </button>
          )}
        </div>
      </div>
    );
  };

  const navItems: BottomNavItem[] = [
    {
      key: 'ofertas',
      label: 'Ofertas',
      badge: ofertas.length > 0 ? ofertas.length : undefined,
      icon: (
        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
            d="M20 7l-8-4-8 4m16 0l-8 4m8-4v10l-8 4m0-10L4 7m8 4v10M4 7v10l8 4"/>
        </svg>
      ),
    },
    {
      key: 'viagens',
      label: 'Viagens',
      badge: viagens.length > 0 ? viagens.length : undefined,
      icon: (
        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
            d="M8 7h12m0 0l-4-4m4 4l-4 4m0 6H4m0 0l4 4m-4-4l4-4"/>
        </svg>
      ),
    },
    {
      key: 'historico',
      label: 'Histórico',
      icon: (
        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
            d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z"/>
        </svg>
      ),
    },
  ];

  const currentList = tab === 'ofertas' ? ofertas : tab === 'viagens' ? viagens : historico;
  const emptyMessages: Record<Tab, string> = {
    ofertas: 'Nenhuma oferta disponível no momento',
    viagens: 'Nenhuma viagem em andamento',
    historico: 'Nenhuma entrega concluída',
  };

  return (
    <div className="min-h-screen bg-slate-50 dark:bg-slate-950 pb-20">
      {/* Header */}
      <header className="sticky top-0 z-30 bg-white dark:bg-slate-900 border-b border-slate-200 dark:border-slate-800 px-4 py-3 flex items-center justify-between">
        <div>
          <h1 className="font-bold text-slate-900 dark:text-white text-base">TSM Link</h1>
          <p className="text-xs text-slate-500 dark:text-slate-400">{user?.email}</p>
        </div>
        <button
          type="button"
          onClick={logout}
          className="text-slate-500 dark:text-slate-400 p-2 rounded-lg"
        >
          <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
              d="M17 16l4-4m0 0l-4-4m4 4H7m6 4v1a3 3 0 01-3 3H6a3 3 0 01-3-3V7a3 3 0 013-3h4a3 3 0 013 3v1"/>
          </svg>
        </button>
      </header>

      <main className="p-4">
        {loading ? (
          <div className="flex items-center justify-center py-20">
            <div className="w-8 h-8 border-2 border-indigo-500 border-t-transparent rounded-full animate-spin"/>
          </div>
        ) : error ? (
          <div className="p-4 bg-red-50 dark:bg-red-900/20 rounded-xl text-red-700 text-sm">
            {error}
            <button onClick={load} className="ml-2 underline">Tentar novamente</button>
          </div>
        ) : currentList.length === 0 ? (
          <div className="text-center py-20 text-slate-400 text-sm">
            {emptyMessages[tab]}
          </div>
        ) : (
          <div className="space-y-3">
            {currentList.map(m => (
              <ManifestCard
                key={m.id}
                manifest={m}
                showActions={tab !== 'historico'}
              />
            ))}
          </div>
        )}
      </main>

      <BottomNav items={navItems} active={tab} onChange={(k) => setTab(k as Tab)} />
    </div>
  );
};

export default LinkHome;
```
